void setup()
{
  pinMode(11, OUTPUT);
  pinMode(8, OUTPUT);
  pinMode(3, OUTPUT);
}

void loop()
{
  digitalWrite(11, HIGH);
  delay(4000);
  digitalWrite(11, LOW);
  
  digitalWrite(8, HIGH);
  delay(2000);
  digitalWrite(8, LOW);
  
  digitalWrite(3, HIGH);
  delay(5000);
  digitalWrite(3, LOW);
}
